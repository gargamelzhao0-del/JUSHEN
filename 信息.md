问题分析
位置: tcp_motor_server.c:1386-1446

static osMutexId_t connect_mutex = NULL;
static bool connect_in_progress = false;

根因:
虽然使用了互斥锁保护，但存在以下问题：

静态变量params被重用: 如果前一次连接尚未完全结束，新的连接请求可能覆盖params结构体
信号量生命周期管理不清晰: params.done_sem在超时时可能未被正确清理
connect_in_progress标志在异常路径中可能未被重置
竞态窗口:


Task A: osMutexAcquire() -> OK
Task A: connect_in_progress = true
Task A: osMutexRelease()  <- 释放锁
    [窗口] Task B: osMutexAcquire() -> 等待或超时
Task A: 继续执行连接...
    [窗口] Task B获取锁，检查connect_in_progress == true，返回false
    [问题] 如果Task A在超时后重置了params_in_use，Task B可能看到不一致状态
最优解决方案

/**
 * @brief TCP客户端连接请求结构（动态分配，避免静态变量竞态）
 */
typedef struct {
    ip_addr_t remote_addr;
    uint16_t remote_port;
    err_t result;
    osSemaphoreId_t done_sem;
    bool in_use;              /* 是否正在使用 */
} TcpConnectRequest_t;

/* 使用动态内存池或请求队列 */
#define MAX_CONNECT_REQUESTS 2
static TcpConnectRequest_t connect_requests[MAX_CONNECT_REQUESTS];
static osMutexId_t connect_pool_mutex = NULL;

/**
 * @brief 分配连接请求结构
 */
static TcpConnectRequest_t* alloc_connect_request(void)
{
    if (connect_pool_mutex == NULL) {
        connect_pool_mutex = osMutexNew(NULL);
    }
    
    osMutexAcquire(connect_pool_mutex, portMAX_DELAY);
    
    for (int i = 0; i < MAX_CONNECT_REQUESTS; i++) {
        if (!connect_requests[i].in_use) {
            connect_requests[i].in_use = true;
            osMutexRelease(connect_pool_mutex);
            return &connect_requests[i];
        }
    }
    
    osMutexRelease(connect_pool_mutex);
    return NULL;
}

/**
 * @brief 释放连接请求结构
 */
static void free_connect_request(TcpConnectRequest_t* req)
{
    if (req == NULL) return;
    
    osMutexAcquire(connect_pool_mutex, portMAX_DELAY);
    req->in_use = false;
    if (req->done_sem != NULL) {
        osSemaphoreDelete(req->done_sem);
        req->done_sem = NULL;
    }
    osMutexRelease(connect_pool_mutex);
}

bool tcp_client_connect(const char* remote_ip, uint16_t remote_port)
{
    /* 分配请求结构 */
    TcpConnectRequest_t* req = alloc_connect_request();
    if (req == NULL) {
        printf("[TCP Client] No available connect request slot\n");
        return false;
    }
    
    /* 初始化请求 */
    req->done_sem = osSemaphoreNew(1, 0, NULL);
    if (req->done_sem == NULL) {
        free_connect_request(req);
        return false;
    }
    
    if (!ip4addr_aton(remote_ip, &req->remote_addr)) {
        free_connect_request(req);
        return false;
    }
    req->remote_port = remote_port;
    req->result = ERR_VAL;
    
    /* 执行连接 */
    err_t err = tcpip_callback_with_block(tcp_client_connect_tcpip_thread, req, 1);
    if (err != ERR_OK) {
        free_connect_request(req);
        return false;
    }
    
    /* 等待完成 */
    osStatus_t status = osSemaphoreAcquire(req->done_sem, 3000);
    if (status != osOK) {
        printf("[TCP Client] Connection timeout\n");
        free_connect_request(req);
        return false;
    }
    
    err_t result = req->result;
    free_connect_request(req);
    
    return (result == ERR_OK);
}

问题分析

TOCTOU (Time-Of-Check-To-Time-Of-Use) 竞态:

client_pcb在互斥锁保护下被检查
但LwIP的错误回调tcp_client_error_cb可能在任何时刻被LwIP调用
错误回调中client_pcb = NULL的操作虽然有线程安全考虑，但与主线程的访问存在竞态
根本原因: client_pcb是一个裸指针，其生命周期由LwIP管理，但代码假设它在我们检查后仍然有效。

最优解决方案
方案A: 使用原子指针操作（推荐）


/* 在tcp_motor_server.c顶部添加 */

static _Atomic(struct tcp_pcb*) client_pcb_atomic = NULL;

    atomic_store(&client_pcb_atomic, pcb);
}

/* 获取PCB时 */
static struct tcp_pcb* get_client_pcb(void) {
    return atomic_load(&client_pcb_atomic);

bool tcp_client_send_data(uint8_t* data, uint16_t len)
{
    if (tcp_client_mutex == NULL || data == NULL || len == 0) return false;
    if (len > TCP_SEND_BUFFER_SIZE) return false;

    osMutexAcquire(tcp_client_mutex, portMAX_DELAY);

    /* 原子获取PCB指针 */
    struct tcp_pcb* pcb = get_client_pcb();
    
    if (!client_connected || pcb == NULL) {
        osMutexRelease(tcp_client_mutex);
        return false;
    }

    /* 使用局部变量pcb，避免多次访问全局变量 */
    if (pcb->state != ESTABLISHED) {
        printf("[TCP] PCB state not ESTABLISHED: %d\n", pcb->state);
        osMutexRelease(tcp_client_mutex);
        return false;
    }

    u16_t snd_buf = tcp_sndbuf(pcb);
    if (snd_buf < len) {
        /* 缓冲区不足处理 */
    }

    memcpy(tcp_send_buffer, data, len);
    tcp_send_len = len;

    /* 传递局部变量而非全局变量 */
    err_t err = tcpip_try_callback(tcp_client_send_tcpip_thread_safe, pcb);
    
    osMutexRelease(tcp_client_mutex);
    return (err == ERR_OK);
}

/* 修改发送回调，使用传递的PCB指针 */
static void tcp_client_send_tcpip_thread_safe(void *arg)
{
    struct tcp_pcb* pcb = (struct tcp_pcb*)arg;
    
    if (pcb == NULL || pcb->state != ESTABLISHED) {
        tcp_async_fail_count++;
        return;
    }
    
    err_t err = tcp_write(pcb, tcp_send_buffer, tcp_send_len, TCP_WRITE_FLAG_COPY);
    if (err == ERR_OK) {
        tcp_output(pcb);
        total_tx_packets++;
        tcp_async_send_count++;
    } else {
        total_tx_errors++;
        tcp_async_fail_count++;
    }
}

{
    osMutexAcquire(tcp_client_mutex, portMAX_DELAY);

        g_tcp_ctx.ref_count++;

        g_tcp_ctx.ref_count--;

问题:

self.rx_buffer.pop(0)对bytearray是O(n)复杂度，每次从头部删除需要移动所有后续数据
多次切片操作self.rx_buffer[0:2]产生临时对象

最优解决方案

# 使用memoryview和索引优化，避免数据拷贝

class QuadrupedTCPClient:
    def __init__(self, host: str = '0.0.0.0', port: int = 9999):
        # ... existing code ...
        
        # V3.4: 优化接收缓冲区处理
        self._rx_buffer = bytearray(65536)  # 预分配缓冲区
        self._rx_head = 0  # 数据起始位置
        self._rx_tail = 0  # 数据结束位置
        self._rx_size = 0  # 当前数据大小
        
        # 帧大小配置（可配置）
        self._frame_sizes = {
            0x02: 24,   # 足端力
            0x03: 200,  # 电机反馈
            0x04: 23,   # 增强电机状态
        }
        
    def _receive_loop(self):
        """接收数据循环 - 优化版"""
        while self.running and self.connected:
            try:
                # 直接写入缓冲区尾部
                write_space = len(self._rx_buffer) - self._rx_tail
                if write_space < 4096:
                    # 缓冲区尾部空间不足，压缩缓冲区
                    self._compact_buffer()
                
                chunk = self.client_socket.recv(min(4096, write_space))
                if not chunk:
                    self._handle_disconnect()
                    break
                
                # 写入缓冲区
                self._rx_buffer[self._rx_tail:self._rx_tail + len(chunk)] = chunk
                self._rx_tail += len(chunk)
                self._rx_size += len(chunk)
                
                self.stats.bytes_received += len(chunk)
                self._process_buffer_optimized()
                
            except socket.timeout:
                continue
            except Exception as e:
                if self.running:
                    self._handle_disconnect()
                break
    
    def _compact_buffer(self):
        """压缩缓冲区，将有效数据移动到头部"""
        if self._rx_head > 0:
            valid_data = self._rx_buffer[self._rx_head:self._rx_tail]
            self._rx_buffer[0:len(valid_data)] = valid_data
            self._rx_tail = len(valid_data)
            self._rx_head = 0
    
    def _process_buffer_optimized(self):
        """优化版帧处理 - 使用索引而非pop"""
        while self._rx_size >= 4:
            # 使用memoryview避免数据拷贝
            view = memoryview(self._rx_buffer)[self._rx_head:self._rx_tail]
            
            # 检查帧头
            if view[0] != 0xAA or view[1] != 0x55:
                self._rx_head += 1
                self._rx_size -= 1
                continue
            
            frame_type = view[2]
            
            # 获取帧大小
            frame_size = self._frame_sizes.get(frame_type)
            if frame_size is None:
                # 未知帧类型，跳过帧头
                self._rx_head += 2
                self._rx_size -= 2
                continue
            
            if self._rx_size < frame_size:
                break  # 数据不完整，等待更多数据
            
            # 提取完整帧
            frame = bytes(view[:frame_size])
            self._rx_head += frame_size
            self._rx_size -= frame_size
            
            # 分发帧处理
            if frame_type == 0x02:
                self._handle_foot_force_frame(frame)
            elif frame_type == 0x03:
                self._handle_motor_feedback_frame(frame)
            elif frame_type == 0x04:
                self._handle_enhanced_motor_status_frame(frame)
        
        # 定期压缩缓冲区
        if self._rx_head > 32768:
            self._compact_buffer()

    uint32_t missed_count;          /* 丢失计数 */
    bool pong_received;             /* 是否收到PONG */
    uint32_t last_pong_tick;        /* 上次PONG时间 */
    bool startup_grace_active;      /* 启动宽限期 */
} HeartbeatHealth_t;

static HeartbeatHealth_t g_hb_health = {0};

/**
 * @brief 更新心跳状态（在收到PONG时调用）
 */
void tcp_on_pong_received(void)
{
    g_hb_health.pong_received = true;
    g_hb_health.last_pong_tick = HAL_GetTick();
    g_hb_health.missed_count = 0;
}

/**
 * @brief 周期性心跳检测（在定时任务中调用，固定1Hz）
 */
bool tcp_heartbeat_check_periodic(void)
{
    uint32_t now = HAL_GetTick();
    
    /* 启动宽限期：5秒内不检测 */
    if (g_hb_health.first_heartbeat_tick == 0) {
        g_hb_health.first_heartbeat_tick = now;
        return true;
    }
    if (now - g_hb_health.first_heartbeat_tick < 5000) {
        return true;
    }
    
    /* 检查PONG超时 */
    if (!g_hb_health.pong_received) {
        g_hb_health.missed_count++;
        LOG_WARN("[TCP] PONG not received, missed_count=%d", g_hb_health.missed_count);
    }
    
    /* 重置PONG标志，等待下一次 */
    g_hb_health.pong_received = false;
    
    return (g_hb_health.missed_count < TCP_HEARTBEAT_MAX_MISSED);
}

/**
 * @brief 快速查询心跳状态（无副作用）
 */
bool tcp_is_heartbeat_healthy(void)
{
    return (g_hb_health.missed_count < TCP_HEARTBEAT_MAX_MISSED);
}
